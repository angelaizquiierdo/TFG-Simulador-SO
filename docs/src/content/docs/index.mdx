import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';
 
## Qué puedes hacer
 
<CardGrid>
  <Card title="Ver la ejecución paso a paso" icon="seti:clock">
    Reproduce la simulación tick a tick y sigue en el diagrama de Gantt qué proceso está en CPU y cuáles esperan.
  </Card>
  <Card title="Editar el escenario" icon="pencil">
    Cambia los procesos (llegada, ráfaga, prioridad) y los parámetros del algoritmo, como el quantum, sin salir de la página.
  </Card>
  <Card title="Comparar dos escenarios" icon="random">
    Ejecuta otro algoritmo o cambia un parámetro sobre los mismos procesos y compara las métricas de ambos.
  </Card>
  <Card title="Consultar las métricas" icon="document">
    Al terminar la simulación, consulta el tiempo de espera, de retorno y de respuesta de cada proceso, y las medias.
  </Card>
</CardGrid>
